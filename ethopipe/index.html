# Type-strict biological telemetry constraint filter
from pydantic import BaseModel, Field, ConfigDict, field_validator
from typing import Literal, Optional
from datetime import datetime

class CanineObservationRecord(BaseModel):
    model_config = ConfigDict(
        strict=True,
        extra='forbid'
    )

    individual_id: str = Field(..., pattern=r"^CANIS-[A-Z0-9]{6}$")
    event_timestamp: datetime
    ethogram_token: Literal[
        "play_bow", "displacement_pant",
        "lip_lick", "direct_gaze", "micro_freeze"
    ]
    heart_rate_bpm: Optional[int] = Field(None, ge=30, le=250)
    basis_of_record: Literal["MachineObservation", "HumanObservation"]

    @field_validator("heart_rate_bpm")
    def validate_merck_canine_bounds(cls, value: Optional[int]):
        if value is not None and not (30 <= value <= 250):
            raise ValueError("Physiological anomaly: BPM violates Merck boundaries.")
        return value
